cmake_minimum_required(VERSION 3.21)
project(ApexMatch VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(APEX_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(APEX_TSAN "Enable ThreadSanitizer (separate from ASan)" OFF)
option(APEX_BUILD_BENCHMARKS "Build benchmark executable" ON)
if(APEX_SANITIZERS AND APEX_TSAN)
    message(FATAL_ERROR "ASan/UBSan and TSan require separate builds")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Use GCC or Clang with C++20 support")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wshadow)
if(APEX_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    add_link_options(-fsanitize=address,undefined)
elseif(APEX_TSAN)
    if(WIN32)
        message(FATAL_ERROR "Run TSan in Linux/WSL2")
    endif()
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(apexmatch src/engine.cpp src/protocol.cpp)
target_include_directories(apexmatch PUBLIC include)
add_library(apex_extensions src/extensions.cpp)
target_link_libraries(apex_extensions PUBLIC apexmatch Threads::Threads)
if(WIN32)
    target_link_libraries(apex_extensions PUBLIC ws2_32)
endif()
add_executable(apex_cli apps/cli.cpp)
target_link_libraries(apex_cli PRIVATE apexmatch)
foreach(app IN ITEMS pipeline server client twap)
    add_executable(apex_${app} apps/${app}.cpp)
    target_link_libraries(apex_${app} PRIVATE apex_extensions)
endforeach()
if(APEX_BUILD_BENCHMARKS)
    add_executable(apexmatch_bench benchmarks/benchmark.cpp)
    target_link_libraries(apexmatch_bench PRIVATE apexmatch)
endif()
enable_testing()
add_executable(apex_tests tests/test_main.cpp tests/core_tests.cpp tests/extensions_tests.cpp)
target_link_libraries(apex_tests PRIVATE apex_extensions)
add_test(NAME core_and_extensions COMMAND apex_tests)
add_test(NAME replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_test.py $<TARGET_FILE:apex_cli> ${CMAKE_CURRENT_SOURCE_DIR}/examples)
add_test(NAME network COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py --server $<TARGET_FILE:apex_server> --client $<TARGET_FILE:apex_client>)
add_test(NAME extension_examples COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/extensions_e2e.py --pipeline $<TARGET_FILE:apex_pipeline> --twap $<TARGET_FILE:apex_twap> --examples ${CMAKE_CURRENT_SOURCE_DIR}/examples)
set_tests_properties(core_and_extensions replay network extension_examples PROPERTIES TIMEOUT 120)
